\begin{theorem}[Variance Boundedness Under Exclusion Masking]
Let $W_{\text{raw}} \sim \mathcal{N}(0, \sigma^2)$ with $\sigma = 0.02$. Let $g_i \in [0, 9]$ be the deterministic exclusion gate multiplier. The variance of the initialized weight parameters $\text{Var}(W_{\text{TTI}})$ is strictly bounded by:
\begin{equation}
    \text{Var}(W_{\text{TTI}}) \le \frac{81}{10^6} \cdot \sigma^2 = 3.24 \times 10^{-8}
\end{equation}
\end{theorem}

\begin{proof}
Let $w_i = w_{\text{raw}, i} \cdot \frac{g_i}{1000}$. Since $w_{\text{raw}, i}$ is zero-mean Gaussian and $g_i$ is a deterministic function of index $i$:
\begin{equation}
    \mathbb{E}[w_i] = \frac{g_i}{1000} \mathbb{E}[w_{\text{raw}, i}] = 0
\end{equation}
The variance of $w_i$ is given by:
\begin{equation}
    \text{Var}(w_i) = \mathbb{E}[w_i^2] = \left(\frac{g_i}{1000}\right)^2 \mathbb{E}[w_{\text{raw}, i}^2] = \left(\frac{g_i}{1000}\right)^2 \sigma^2
\end{equation}
Because the exclusion gate output satisfies $0 \le g_i \le 9$, we have $g_i^2 \le 81$.
Substituting bounds yields:
\begin{equation}
    \text{Var}(w_i) \le \frac{81}{10^6} \cdot (0.02)^2 = 8.1 \times 10^{-5} \cdot 4 \times 10^{-4} = 3.24 \times 10^{-8}
\end{equation}
Thus, weight parameters remain strictly bounded, preventing gradient explosion in deep architectures.
\end{proof}
