\section{Formal Mathematical Proofs}

\begin{theorem}[Contraction and Convergence of Shard Accumulation]
Let $S_k \in [0, 1)^D$ be the sequence of folded shards generated by \texttt{PhiInfinityShardFolding}. The sequence of partial sums $C_K(x) = \sum_{k=1}^K S_k \phi^{-(k-1)}$ converges uniformly as $K \to \infty$, with truncation error strictly bounded by $\frac{\phi^{-(K-1)}}{\phi - 1}$.
\end{theorem}

\begin{proof}
Since $S_k = \frac{\text{fmod}(\lfloor \dots \rfloor, M)}{M} \in [0, 1)$, we have $\|S_k\|_\infty \le 1$ for all $k \ge 1$.
The remainder term $R_K = \sum_{k=K+1}^\infty S_k \phi^{-(k-1)}$ satisfies:
\begin{equation}
    \|R_K\|_\infty \le \sum_{k=K+1}^\infty \|S_k\|_\infty \phi^{-(k-1)} \le \sum_{j=K}^\infty \phi^{-j}
\end{equation}
Using the geometric series formula for ratio $r = \phi^{-1} = \frac{\sqrt{5}-1}{2} \approx 0.618034 < 1$:
\begin{equation}
    \sum_{j=K}^\infty \phi^{-j} = \phi^{-K} \cdot \frac{1}{1 - \phi^{-1}} = \phi^{-K} \cdot \frac{\phi}{\phi - 1} = \frac{\phi^{-(K-1)}}{\phi - 1}
\end{equation}
For $K = 3$, $\frac{\phi^{-2}}{\phi - 1} = \frac{0.381966}{0.618034} \approx 0.618034$. For $K = 10$, the truncation error falls below $1.4 \times 10^{-2}$, establishing geometric convergence.
\end{proof}

\begin{theorem}[Sign Conservation Invariance]
For any input vector $x \in \mathbb{R}^D$, $\text{sign}(\texttt{PhiInfinityShardFolding}(x)) = \text{sign}(x)$ pointwise.
\end{theorem}

\begin{proof}
Let $x_i \in \mathbb{R}$ be an arbitrary scalar component. The mantissa $m_i = \text{frac}(|x_i|) \ge 0$. The QRT response $\mathcal{Q}(m_i) \ge 0$, and the virtual modulus operation $\text{fmod}(\lfloor \text{fold\_val} \cdot M \rceil, M) \ge 0$. Thus, $S_{k, i} \ge 0$ for all $k$.
The sum $\sum_{k=1}^{k_{\text{steps}}} S_{k, i} \phi^{-(k-1)} \ge 0$.
Multiplying by $s_i = \text{sign}(x_i)$ yields $\hat{x}_i = s_i \cdot (\text{non-negative value})$.
Therefore, $\text{sign}(\hat{x}_i) = \text{sign}(x_i)$, proving exact sign preservation.
\end{proof}
