\section{Architecture \& Implementation}
The \texttt{PhiInfinityShardFolding} module is implemented in PyTorch in \texttt{nrc\_ai/shard\_folding.py}.

\subsection{PyTorch Implementation}
\begin{lstlisting}[language=Python]
import math
import torch
import torch.nn as nn
from nrc.math import qrt_damping

PHI: float = (1.0 + math.sqrt(5.0)) / 2.0

class PhiInfinityShardFolding(nn.Module):
    def __init__(self, k_steps: int = 3, virtual_modulus: float = 1e8) -> None:
        super().__init__()
        self.k_steps = k_steps
        self.virtual_modulus = virtual_modulus

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        mantissa = torch.frac(torch.abs(x))
        signs = torch.sign(x)

        compressed = torch.zeros_like(x)
        damping_factor = 1.0

        for k in range(1, self.k_steps + 1):
            qrt_active = qrt_damping(mantissa)
            assert isinstance(qrt_active, torch.Tensor)

            phi_pow = PHI ** (6 * k)
            fold_val = qrt_active * phi_pow

            shard_k = torch.fmod(torch.round(fold_val * self.virtual_modulus), self.virtual_modulus)
            shard_k = shard_k / self.virtual_modulus

            compressed += shard_k * (1.0 / damping_factor)
            damping_factor *= PHI

        return compressed * signs
\end{lstlisting}

\subsection{Integration in NRC Stack}
\texttt{PhiInfinityShardFolding} serves as the primary compression engine inside \texttt{ResonanceShardKVCache}, compressing Key and Value tensors whenever sequence token limits are exceeded.
